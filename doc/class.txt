类图

--------------------------------------------------
|                  CImgView
|                负责显示图像
--------------------------------------------------
| - INTERVAL_CHECK_IMG_LOADED_MS: int = 50 {检查图像是否载入完成的时间间隔}
| - m_checkLoaded: QTimer * = nullptr {每隔一段时间检查图片是否加载完成}
| - m_imgMgr: CImgMgr * = nullptr
| - m_displayPolicy : CDisplayPolicy * = nullptr
| - m_displaySt: TDisplaySt
--------------------------------------------------
| + init(imgPath:QString) : int
| + display(imgPath) : int
| + displayPrev(void) : int
| + displayNext(void) : int
| - display(img:QPixmap *) : int
| - display(imgFile: pImgFile_t) : int
| - moveVisableArea(dx:int, dy:int) : int
| - zoom(percent:int) : int {percent = 1~1000}
| - updateIfLoaded(void) : int
--------------------------------------------------

--------------------------------------------------
|                CImgMgr
|   抽象类, 子类有CCacheByNum, CCacheAll
|      负责管理图片缓存与切换顺序
--------------------------------------------------
| ~ m_imgLoaderThr: QThread {读图动作单开一个线程}
--------------------------------------------------
| + init(imgPath: QString, fileMgr: CFileMgr*) : int
| + cur(void) : pImgFile_t
| + prev(void) : pImgFile_t
| + next(void) : pImgFile_t
--------------------------------------------------

--------------------------------------------------
|              CImgLoader
|             负责读取图片
--------------------------------------------------
| + onSigLoadImg(pImgFile: var) : void { 信号槽 }
--------------------------------------------------

--------------------------------------------------
|                CFileMgr
|      负责处理路径下的读取顺序, 
|        维护文件信息和相对次序
--------------------------------------------------
| - m_fileList: QFileInfoList
--------------------------------------------------
| + init(imgPath: QString) : int
| + size(void) : QFileInfo::size_type
| + fileBefore(file: QFileInfo) : QFileInfo {获取指定文件前的文件}
| + nFilesBefore(n: int, file: QFileInfo) : QFileInfoList {获取指定文件前的n个文件}
| + fileAfter(file: QFileInfo) : QFileInfo {获取指定文件后的文件}
| + nFilesAfter(n: int, file: QFileInfo) : QFileInfoList {获取指定文件后的n个文件}
| + allFiles(void) : QFileInfoList
--------------------------------------------------
