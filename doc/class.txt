| - m_dspSt: TDisplaySt

| - m_dspStrategy: CDspStrategy*

| - m_cfgMgr : const CCfgMgr * = nullptr
|-------------------------------------------------
| + init(cfg : const CCFG *) : int
| + setLbImg(lb: QLabel*) : void

| ~ event(ev: QEvent*) : bool {处理ScrollArea默认未处理的事件}
| ~ keyPressEvent(ev: QKeyEvent*) : void 
| ~ wheelEvent(ev: QWheelEvent*) : void
| ~ resizeEvent(ev: QResizeEvent*) : void
| - updateImgFocus(void) : void {更新图片焦点坐标}
| - diff_topLeft2Center(void) : QPointF {显示区域左上角到中央的距离}
| - moveSight(handler: EV_HANDLER) : int
| - moveSight(dx, dy) : int
| - zoom(handler: EV_HANDLER) : int
| - zoom(int percent) : int
| - dspWidth(void) : int {dspArea的宽高(不包括边框和滚动条)}
| - dspHeight(void) : int
