类图

 -------------------------------------------------
|                  CImgView
|             负责显示和切换图像
|-------------------------------------------------
| - INTERVAL_CHECK_IMG_LOADED_MS: int = 50 {检查图像是否载入完成的时间间隔}
| - m_checkLoaded: QTimer * = nullptr {每隔一段时间检查图片是否加载完成}
| - m_imgMgr: CImgMgr * = nullptr
|-------------------------------------------------
| + init(filePath: QString) : int
| + displayPrev(void) : int
| + displayNext(void) : int
| - display(img: QPixmap*) : int
| - display(imgFile: pImgFile_t) : int
| - updateIfLoaded(void) : int
 -------------------------------------------------

 -------------------------------------------------
|                CImgDspArea
|          负责显示, 移动和缩放图像
|-------------------------------------------------
| + DSP_STRATEGY: enum {RealSize, FitWin,};    //图片显示策略
| - ZOOM_MAX: int = 1000 {最大缩放倍率, 单位为%}
| - ZOOM_MIN: int = 25
| - m_curImg: QPixmap
| - m_dspSt: TDisplaySt
| - m_lbImg: QLabel*
|-------------------------------------------------
| + setLbImg(lb: QLabel*) : void
| + display(img: QPixmap*) : int
| ~ event(ev: QEvent*) : bool {处理ScrollArea默认未处理的事件}
| ~ keyPressEvent(ev: QKeyEvent*) : void 
| ~ wheelEvent(ev: QWheelEvent*) : void
| ~ resizeEvent(ev: QResizeEvent*) : void
| - updateImgFocus(void) : void {更新图片焦点坐标}
| - diff_topLeft2Center(void) : QPointF {显示区域左上角到中央的距离}
| - zoom(int percent) : int
| - dspWidth(void) : int {dspArea的宽高(不包括边框和滚动条)}
| - dspHeight(void) : int
 -------------------------------------------------

 -------------------------------------------------
|                CImgMgr
|   抽象类, 子类有CCacheByNum, CCacheAll
|      负责管理图片缓存与切换顺序
|-------------------------------------------------
| ~ m_imgLoaderThr: QThread {读图动作单开一个线程}
|-------------------------------------------------
| + init(filePath: QString, sorting: QDir::sortflags) : int
| + cur(void) : pImgFile_t
| + prev(void) : pImgFile_t
| + next(void) : pImgFile_t
 -------------------------------------------------

 -------------------------------------------------
|              CImgLoader
|             负责读取图片
|-------------------------------------------------
| + onSigLoadImg(pImgFile: var) : void { 信号槽 }
 -------------------------------------------------

 -------------------------------------------------
|                CFileMgr
|      负责处理路径下的读取顺序, 
|        维护文件信息和相对次序
|-------------------------------------------------
| - m_fileList: QFileInfoList
|-------------------------------------------------
| + init(filePath: QString, dir: QDir) : int
| + size(void) : QFileInfo::size_type
| + fileBefore(file: QFileInfo) : QFileInfo {获取指定文件前的文件}
| + nFilesBefore(n: int, file: QFileInfo) : QFileInfoList {获取指定文件前的n个文件}
| + fileAfter(file: QFileInfo) : QFileInfo {获取指定文件后的文件}
| + nFilesAfter(n: int, file: QFileInfo) : QFileInfoList {获取指定文件后的n个文件}
| + allFiles(void) : QFileInfoList
 -------------------------------------------------

 -------------------------------------------------
|                CDspStrategy
|       负责以多种策略处理即将显示的图像
|-------------------------------------------------
| + process(img: QPixmap*) : int {img即是输入参数也是输出参数}
 -------------------------------------------------

 -------------------------------------------------
|                TImgFile

|-------------------------------------------------
| + m_info: QFileInfo
| + m_pImg: QPixmap*

|         当前显示图片的显示状态
