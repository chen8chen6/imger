类图

--------------------------------------------------
|                  CImgView
|                负责显示图像
--------------------------------------------------
| - m_cache: CImgCache * = nullptr
| - m_curImg: QPixmap * = nullptr
| - m_displayPolicy : CDisplayPolicy * = nullptr
| - m_displaySt: TDisplaySt
--------------------------------------------------
| + init(imgPath:QString) : int
| + display(imgPath) : int
| + displayPrev(void) : int
| + displayNext(void) : int
| - display(img:QPixmap *) : int
| - moveVisableArea(dx:int, dy:int) : int
| - zoom(percent:int) : int {percent = 1~1000}
--------------------------------------------------

--------------------------------------------------
|                CImgCache
|               负责缓存图像
--------------------------------------------------
| - m_fileMgr: CFileMgr* = nullptr
| - m_buf: CCacheBuffer* = nullptr
--------------------------------------------------
| + init(imgPath: QString) : int
| + cur(void) : TImgFile
| + prev(void) : TImgFile
| + next(void) : TImgFile
--------------------------------------------------

--------------------------------------------------
|                CFileMgr
|      负责处理路径下的读取顺序, 
|        维护文件信息和相对次序
--------------------------------------------------
| - m_fileList: QFileInfoList
| - m_prev: iterator
| - m_next: iterator
--------------------------------------------------
| + init(imgPath: QString) : int
| + prev(void) : QFileInfo
| + next(void) : QFileInfo
| + size(void) : int
--------------------------------------------------

--------------------------------------------------
|                CCacheBuffer
|           负责简化处理缓存管理
--------------------------------------------------
| + BUF_SIZE: const int
| - m_cache: vector<TImgFile> {size = BUF_SIZE}
| - m_idxBegin: int
| - m_idxEnd: int
| - m_idxCur: int
--------------------------------------------------
| + init(imgPath: QString) : int
| + cache(imgPath:QString) : int
| + reset(void) : void
--------------------------------------------------

--------------------------------------------------
|                CDisplayPolicy
|       负责以多种策略处理即将显示的图像
--------------------------------------------------
| + process(img: QPixmap&) : int
--------------------------------------------------

--------------------------------------------------
|                TImgFile
|         文件信息和已载入的图像数据
--------------------------------------------------
| + m_info: QFileInfo
| + m_img: QPixmap*
--------------------------------------------------

--------------------------------------------------
|                TDisplaySt
|         当前显示图片的显示状态
--------------------------------------------------
| + m_focusX: int {屏幕中央在图片中的坐标}
| + m_focusY: int
| + m_zoomPercent: int {放大倍率}
--------------------------------------------------
